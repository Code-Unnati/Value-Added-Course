CLASS zcl_loops_demo DEFINITION
  PUBLIC
  FINAL
  CREATE PUBLIC.

  PUBLIC SECTION.
    INTERFACES if_oo_adt_classrun.

ENDCLASS.

CLASS zcl_loops_demo IMPLEMENTATION.

  METHOD if_oo_adt_classrun~main.

    out->write( 'DO Loop:' ).
    DO 5 TIMES.
      out->write( |Iteration: { sy-index }| ).
    ENDDO.

    out->write( 'WHILE Loop:' ).
    DATA(lv_counter) = 1.
    WHILE lv_counter <= 5.
      out->write( |Counter: { lv_counter }| ).
      lv_counter = lv_counter + 1.
    ENDWHILE.

    out->write( 'LOOP AT (internal table):' ).
    DATA(lt_numbers) = VALUE int4_table( ( 10 ) ( 20 ) ( 30 ) ).
    LOOP AT lt_numbers INTO DATA(lv_val).
      out->write( |Value: { lv_val }| ).
    ENDLOOP.

  ENDMETHOD.

ENDCLASS.
